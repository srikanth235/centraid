      text #field/memo "Memo"  rev=0
      choice #field/status "Status"

        option "Pending"
      repeat "Splits" #repeat/splits
        add "Add split" sym="plus" #repeat/splits/add
      explicit save="Save"#form/save cancel="Cancel"#form/cancel save_disabled
